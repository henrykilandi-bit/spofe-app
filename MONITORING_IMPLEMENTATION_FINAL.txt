🎯 ÉTAPES D'INTÉGRATION (COPIER/COLLER)
═══════════════════════════════════════════════════════════════════════════════

ÉTAPE 1: Installer dépendances
────────────────────────────────────────────────────────────────────────────
npm install prom-client node-cron


ÉTAPE 2: Mettre à jour .env
────────────────────────────────────────────────────────────────────────────
# Ajouter ces lignes:
NOTIFICATION_SLACK=true
SLACK_WEBHOOK_URL=https://hooks.slack.com/services/YOUR/WEBHOOK/URL
PROMETHEUS_ENABLED=true
PROMETHEUS_SCRAPE_INTERVAL=30000


ÉTAPE 3: Intégrer dans cascade/src/app.js
────────────────────────────────────────────────────────────────────────────

# Au début du fichier, après les imports existants:
import {
  initializeMonitoring,
  prometheusMiddleware,
  errorNotificationMiddleware,
  gracefulShutdown
} from './monitoring/bootstrap.js';

# Dans la fonction setupMiddleware() ou au démarrage:
// 1. Ajouter middleware Prometheus (AVANT les routes)
prometheusMiddleware(app);

// 2. Ajouter middleware notifications d'erreur (AVANT les routes)
errorNotificationMiddleware(app);

// 3. Initialiser le monitoring (APRÈS tous les autres middlewares)
const { supervisor, prometheus, notifier } = await initializeMonitoring(app);

// Les routes health-check sont automatiquement attachées


ÉTAPE 4: Ajouter graceful shutdown dans cascade/src/server.js
────────────────────────────────────────────────────────────────────────────

# À la fin du fichier, après app.listen():
import { gracefulShutdown } from './monitoring/bootstrap.js';

const server = app.listen(PORT);

// Graceful shutdown
process.on('SIGTERM', gracefulShutdown(app, server));
process.on('SIGINT', gracefulShutdown(app, server));


ÉTAPE 5: Démarrer et tester
────────────────────────────────────────────────────────────────────────────
npm run dev

# Dans un autre terminal:
curl http://localhost:3001/api/health/status
curl http://localhost:3001/metrics

Dans votre code - Utiliser les services:
───────────────────────────────────────────────────────────────────────────
import NotificationService from './src/utils/notification-service.js';
import PrometheusService from './src/utils/prometheus-service.js';

// Obtenir les instances
const notifier = NotificationService.getInstance();
const prometheus = PrometheusService.getInstance();

// Envoyer alerte critique
await notifier.sendCriticalAlert('Database Down', {
  component: 'database',
  error: 'Connection lost'
});

// Enregistrer métrique
prometheus.recordDatabaseQueryTime('SELECT', 150);
prometheus.recordAPIRequest('GET', '/api/users', 200);
